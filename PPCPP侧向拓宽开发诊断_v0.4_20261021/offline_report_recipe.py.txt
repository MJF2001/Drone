    with Path(p).open('x',encoding='utf8') as f:json.dump(x,f,ensure_ascii=False,indent=2)

    with Path(p).open('x',encoding='utf8') as f:f.write(s)

def diagram(st):

    for k,method in enumerate(('INPUT',)+methods):
        x0=k*panel+12;y0=54;row=None;r=[]
        if method!='INPUT':
            row=next(x for x in rows if x['case']==st['id'] and x['method']==method and x['seed']==20261020);p=H/'outputs'/f'{row["id"]}.json'
            if p.exists():r=read(p)['result']['raw'].get('route',[])
        parts.append(f'<text x="{x0}" y="17">{html.escape(st["id"] if k==0 else method)}</text>')
        if row:
            m=row['metrics'];label=f'L={m["length"]}, C={m["completion_step"]}, R={m["return_tail_steps"]}' if m else str(row['status']);parts.append(f'<text x="{x0}" y="34">{label}; valid={row["valid"]}, timely={row["timely"]}</text>')
        for x,y in sorted(F):
            p=(x,y);color='#ffb64d' if p in A else '#8bb7ec' if p in past else '#ffeaaa' if p in T else '#f5f5f5';parts.append(f'<rect x="{x0+x*s}" y="{y0+y*s}" width="{s}" height="{s}" fill="{color}" stroke="#ccc" stroke-width=".4"/>')
        def pt(p):return (x0+(p[0]+.5)*s,y0+(p[1]+.5)*s)
        if r and row['metrics']:
            C=row['metrics']['completion_step']
            for rr,color,dash in [(r[:C+1],'#16672c',''),(r[C:],'#cc0033','stroke-dasharray="4 2"')]:
                coords=' '.join(f'{pt(p)[0]},{pt(p)[1]}' for p in rr);parts.append(f'<polyline points="{coords}" fill="none" stroke="{color}" stroke-width="2" {dash}/>')
            v=row['visits_A'];first=v['capture_order'][0]['index'] if v and v['capture_order'] else None
            marks=[(first,'#00a6b2','first'),(v.get('first_A_capture') if v else None,'#1283e8','A1'),(v.get('A_completion') if v else None,'#923dd1','Ac'),(C,'#111','C')]
            for index,color,label in marks:
                if index is None:continue
                x,y=pt(r[index]);parts.append(f'<circle cx="{x}" cy="{y}" r="5" fill="white" stroke="{color}" stroke-width="2.5"/><text x="{x+6}" y="{y-6}" fill="{color}">{label}@{index}</text>')
        for p,color,label in [(c['base'],'black','b'),(c['current'],'#e50070','s')]:
            x,y=pt(p);parts.append(f'<circle cx="{x}" cy="{y}" r="3" fill="{color}"/><text x="{x+4}" y="{y+13}" fill="{color}">{label}</text>')
    parts.append(f'<text x="12" y="{height-38}">橙A／蓝H／黄旧待检；s当前／b基地；first首次剩余目标采集，A1首次A采集，Ac完成A，C全部采集完成。</text><text x="12" y="{height-18}">绿线：至C的采集阶段；红虚线：C后返航。固定seed20261020；重叠路径及标记保持实际位置。</text></svg>');name=st['id']+'_annotated.svg';put(d/name,''.join(parts));ET.parse(d/name);return name
