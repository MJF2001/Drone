"""Independent formula/BFS replay. Imports no planner, hook, reward or geometry."""
import math,json,time
from collections import Counter,deque
def norm(x):return json.loads(json.dumps(x))
def ps(xs):return set(map(tuple,xs))
def require(v,msg):
 if not v:raise AssertionError(msg)
def audit(c,native,spec,enabled):
 start=time.perf_counter();counts=Counter(BFS_calls=0,BFS_popped=0,BFS_neighbors=0,unit_edges=0,decisions=0,local_option_rows=0,pending_state_checks=0,authorization_checks=0)
 r=native['raw']['branches']['P0'];diag=native['diagnostic'];route=list(map(tuple,r['route']));F=ps(c['free']);T=ps(c['targets']);HH=ps(c['covered']);events=r['trace'];pred=tuple(c['predator']);pending=T-HH-{tuple(c['current'])};cursor=0;authorized=0;receipts=[]
 def bfs(a):
  counts['BFS_calls']+=1;d={a:0};q=deque([a])
  while q:
   x,y=q.popleft();counts['BFS_popped']+=1
   for p in ((x-1,y),(x+1,y),(x,y-1),(x,y+1)):
    counts['BFS_neighbors']+=1
    if p in F and p not in d:d[p]=d[x,y]+1;q.append(p)
  return d
 require(route and route[0]==tuple(c['current']) and route[-1]==tuple(c['base']),'wrong start/base')
 require(set(route)<=F and T<=HH|set(route) and len(route)-1<=c['distance_budget'],'collision/coverage/resource')
 for a,b in zip(route,route[1:]):counts['unit_edges']+=1;require(sum(abs(x-y) for x,y in zip(a,b))==1,'non-unit edge')
 require(r['status']=='core_complete' and 0<=r['accepted_seconds']<r['budget'] and not r['fallback_selected'],'core acceptance')
 for i,e in enumerate(events):
  counts['decisions']+=1;x,y=route[cursor];prev=route[cursor-1] if cursor else (tuple(c['previous']) if c['previous'] is not None else None)
  require(e['index']==cursor and tuple(e['current'])==(x,y) and e['pending_count']==len(pending),'decision state')
  if e['kind']!='return':
   require((None if e['previous'] is None else tuple(e['previous']))==prev,'heading mismatch')
   record=diag['decision_states'][i];require(record['trace_record']==i and norm(record['event'])==norm(e) and ps(record['pending'])==pending,'full diagnostic pending/event mismatch');counts['pending_state_checks']+=1
  goal=tuple(e['goal']);end=events[i+1]['index'] if i+1<len(events) else len(route)-1
  require(cursor<=end<len(route),'decision interval bounds');segment=route[cursor:end+1];require(segment[-1]==goal,'goal not reached')
  if e['kind']!='return':require(norm(segment)==diag['decision_states'][i]['segment'],'recorded segment mismatch')
  opts=sorted(q for q in ((x-1,y),(x+1,y),(x,y-1),(x,y+1)) if q in pending and q in F);default=None
  if e['kind']=='local':
   require(opts and end==cursor+1,'local rule applicability');ds=[math.dist(q,pred) for q in opts];lo,hi=min(ds),max(ds);rows=[]
   for q,d in zip(opts,ds):
    rd=(d-lo)/(hi-lo) if hi>lo else 0;rs=0 if prev is None else math.acos(max(-1,min(1,(prev[0]-x)*(q[0]-x)+(prev[1]-y)*(q[1]-y))))/math.pi
    rb=1-sum(p in pending and p in F for p in ((q[0]-1,q[1]),(q[0]+1,q[1]),(q[0],q[1]-1),(q[0],q[1]+1)))/4
    rows.append(dict(cell=q,distance_reward=rd,straight_reward=rs,boundary_reward=rb,score=rd+2*rs+.5*rb));counts['local_option_rows']+=1
   require(len(rows)==len(e['options']),'option count')
   for expected,actual in zip(rows,e['options']):
    require(expected['cell']==tuple(actual['cell']),'option cell')
    for k in ('distance_reward','straight_reward','boundary_reward','score'):require(abs(expected[k]-actual[k])<1e-12,'option reward')
   require(goal==min(rows,key=lambda v:(-v['score'],v['cell']))['cell'],'unauthorized local choice')
  elif e['kind']=='recovery':
   require(not opts and pending,'recovery applicability');d=bfs((x,y));require(pending<=set(d),'unreachable pending');default=min(pending,key=lambda p:(d[p],p))
   if i==spec['trace_record']:
    counts['authorization_checks']+=1
    require(norm(events[:i])==spec['original_trace'][:i] and norm(route[:cursor+1])==spec['original_route_prefix'],'authorization prefix')
    require([x,y]==spec['current'] and norm(prev)==spec['previous'] and pending==ps(spec['pending']),'authorization full state')
    require(default==tuple(spec['default_goal']),'authorization default target')
    alt=tuple(spec['alternative_goal']);require(alt in pending and alt in F and alt in d,'authorized target identity/reachability')
    require(goal==(alt if enabled else default),'authorized goal mismatch');authorized+=int(enabled)
    tr=diag['trigger'];require(tr['default_distance']==d[default] and tr['alternative_distance']==d[alt] and ps(tr['pending'])==pending,'diagnostic distances/pending')
    require(tr['route_index']==cursor and tr['trace_record']==i and tr['chosen_goal']==list(goal) and tr['actual_recovery_path']==norm(segment),'diagnostic trigger/path')
   else:require(goal==default,'unauthorized second recovery choice')
   require(goal in pending and end-cursor==d[goal],'recovery shortest/legal goal')
  else:
   require(e['kind']=='return' and not pending and goal==tuple(c['base']),'return rule');require(end-cursor==bfs((x,y))[goal],'return shortest')
  receipts.append({'trace_record':i,'route_index':cursor,'route_end':end,'kind':e['kind'],'previous':norm(prev),'current':[x,y],'pending_before':norm(sorted(pending)),'default_goal':norm(default),'actual_goal':list(goal),'segment_new_tasks':norm(sorted(pending&set(segment))),'authorized_exception':enabled and i==spec['trace_record']})
  pending-=set(segment);cursor=end
 require(cursor==len(route)-1 and not pending and authorized==int(enabled),'final state/exception count')
 require(diag['target_intervention_enabled']==enabled and diag['core_detail_profile_enabled'] is False,'distinct OFF/profile switches')
 require(diag['counts']['matched']==1 and diag['counts']['applied']==int(enabled) and diag['counts']['explanatory_distance_queries']==2,'diagnostic counts')
 require(len(diag['decision_states'])==len(events)-1,'diagnostic record count')
 return {'passed':True,'counts':dict(counts),'authorized_exceptions':authorized,'decisions':receipts,'seconds':time.perf_counter()-start,'scope':'independent reward formulas, all unit edges, goal identities, full pending, heading, FIFO distance fields and all later original rules; one authorized exception only; no coverage planning/rollout','path_tie_scope':'shortestness independently verified; exact original path tie function is bound by unchanged source and original geometry object, not a second tie implementation'}
